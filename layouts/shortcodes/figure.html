{{- /* Custom override of Hugo's built-in "figure" shortcode. Same call syntax as the default shortcode (src, alt, width, caption, class, link, target, rel, title), so no existing post content needs to change. What it does differently: - JPEG/PNG images are resized and a WebP copy (quality 82) is generated alongside the original format, served via so the browser picks whichever it supports. - "width" controls the ON-PAGE display size (the width/height attributes), NOT the pixel resolution of the file that gets served. The actual file is rendered at $displayDensity times that width (default 1.5x) so it stays sharp on high-DPI/retina screens and has a little headroom instead of being served pixel-for-pixel at the display box size. This keeps "width" meaning what it always meant in existing posts (how big the photo looks on the page) while fixing blur. - GIFs and SVGs pass through untouched (animation / vector scaling would break otherwise). - Remote images (http/https src) or anything Hugo can't resolve as a local resource also pass through untouched, so the shortcode never breaks a build on an unexpected src. - loading="lazy" is added to every for off-screen images. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" -}} {{- $caption := .Get "caption" -}} {{- $width := .Get "width" -}} {{- $class := .Get "class" -}} {{- $link := .Get "link" -}} {{- $target := .Get "target" -}} {{- $rel := .Get "rel" -}} {{- $title := .Get "title" -}} {{- $maxWidth := 1600 -}} {{- $ext := path.Ext $src | lower -}} {{- $isRemote := or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://") -}} {{- $skipProcessing := or (eq $ext ".gif") (eq $ext ".svg") $isRemote -}} {{- $image := "" -}} {{- if not $skipProcessing -}} {{- with .Page.Resources.GetMatch $src -}} {{- $image = . -}} {{- else -}} {{- with resources.Get $src -}} {{- $image = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if $link }}{{ end -}} {{- if $image -}} {{- $displayWidth := $maxWidth -}} {{- with $width }}{{ $displayWidth = int . }}{{ end -}} {{- if lt $image.Width $displayWidth }}{{ $displayWidth = $image.Width }}{{ end -}} {{- $displayHeight := div (mul $displayWidth $image.Height) $image.Width -}} {{- $displayDensity := 1.5 -}} {{- $sourceWidth := int (mul $displayWidth $displayDensity) -}} {{- if gt $sourceWidth $image.Width }}{{ $sourceWidth = $image.Width }}{{ end -}} {{- $resized := $image.Resize (printf "%dx" $sourceWidth) -}} {{- $webp := $image.Resize (printf "%dx webp q82" $sourceWidth) -}} {{- else -}} {{- /* GIF, SVG, remote, or unresolvable src: render as-is, no processing */ -}} {{- end -}} {{- if $link }}{{ end -}} {{- with $caption }} {{- end }}
